        {\huge\bfseries}
        {}{0.1em}{}

    % This is a new definition specifically for numbered main matter chapters
    \titlecontents{chapter}[0.5em]
        {\addvspace{1em}\bfseries}
        {Chapter~\thecontentslabel~--~}
        {\bfseries}
        {\hfill\thecontentspage}
    
    \titlecontents{numberedchapter}[0.5em]
        {\addvspace{1em}\bfseries}
        {Chapter \thecontentslabel~--~} % "Chapter X -- "
        {}
        {\hfill\thecontentspage}
        
    \titlecontents{section}[2.2em]
        {\normalfont}
        {\S\thecontentslabel\quad}
        {}
        {\hfill\thecontentspage}
        
    \titlecontents{subsection}[5.4em]
        {\normalfont}
        {\S\S\thecontentslabel\quad}
        {}
        {\hfill\thecontentspage}

% Chapter Heading formatting
    \titleformat{\chapter}[hang]
        {\huge\bfseries}
        {}{0.1em}{}
        [{
            \startcontents[chapter]
            \vspace{1em}\noindent\textbf{\normalsize Contents}
            {\normalfont\small
            \setlength{\parskip}{0pt}
            \printcontents[chapter]{l}{1}[2]{}
            }
        }]
        % Chapter Table of Contents 
        \titlecontents{lsection}[0em]
            {}{\S\thecontentslabel\quad}
            {}{\titlerule*[1pc]{}\thecontentspage}
        \titlecontents{lsubsection}[2em]
            {}{\S\S\thecontentslabel\quad}
            {}{\titlerule*[1pc]{}\thecontentspage}

    \titleformat{\section}
        {\normalfont\bfseries} % Format for the whole heading
        {\S\thesection}              % The label: § + number
        {1em}                        % Horizontal space between label and title
        {}                           % Code before the title (none)
    \titleformat{\subsection}
        {\normalfont\bfseries}
        {\S\S\thesubsection}
        {1em}
        {}
    \titleformat{\subsubsection}
        {\normalfont\bfseries}
        {\S\S\thesubsection-\thesubsubsection}
        {1em}
        {}
